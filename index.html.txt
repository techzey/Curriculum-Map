        const STATUSES = ['not-taken', 'in-progress', 'passed'];
        let courseStatus = JSON.parse(localStorage.getItem('me_curriculum_status')) || {};

        function renderCurriculum() {
            const grid = document.getElementById('curriculum-grid');
            grid.innerHTML = '';

            curriculumData.forEach(termData => {
                const card = document.createElement('div');
                card.className = 'term-card';
                
                let totalTermUnits = termData.courses.reduce((acc, c) => acc + c.units, 0);
                
                card.innerHTML = `<div class="term-title">Year ${termData.year} - Term ${termData.term} (${totalTermUnits} Units)</div>`;

                termData.courses.forEach(course => {
                    const status = courseStatus[course.code] || 'not-taken';
                    const courseEl = document.createElement('div');
                    courseEl.className = `course-item ${status}`;
                    courseEl.dataset.status = status;
                    courseEl.dataset.code = course.code;
                    courseEl.onclick = () => cycleStatus(course.code);

                    courseEl.innerHTML = `
                        <div class="course-header">
                            <span>${course.code}</span>
                            <span class="status-badge">${status.replace('-', ' ')}</span>
                        </div>
                        <div class="course-title">${course.title}</div>
                        <div class="course-meta">Units: ${course.units} | Pre-req: ${course.req}</div>
                    `;
                    card.appendChild(courseEl);
                });

                grid.appendChild(card);
            });

            updateStats();
        }

        function cycleStatus(code) {
            const current = courseStatus[code] || 'not-taken';
            const nextIndex = (STATUSES.indexOf(current) + 1) % STATUSES.length;
            courseStatus[code] = STATUSES[nextIndex];
            
            localStorage.setItem('me_curriculum_status', JSON.stringify(courseStatus));
            renderCurriculum();
        }

        function updateStats() {
            let completedUnits = 0;
            const TOTAL_UNITS = 210;

            curriculumData.forEach(term => {
                term.courses.forEach(course => {
                    if (courseStatus[course.code] === 'passed') {
                        completedUnits += course.units;
                    }
                });
            });

            const remainingUnits = TOTAL_UNITS - completedUnits;
            const percentage = Math.round((completedUnits / TOTAL_UNITS) * 100);

            document.getElementById('completed-units').innerText = completedUnits;
            document.getElementById('remaining-units').innerText = remainingUnits;
            document.getElementById('progress-percent').innerText = `${percentage}%`;
            document.getElementById('progress-fill').style.width = `${percentage}%`;
        }

        function filterCourses(filter, btn) {
            document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');

            const items = document.querySelectorAll('.course-item');
            items.forEach(item => {
                if (filter === 'all' || item.classList.contains(filter)) {
                    item.style.display = 'block';
                } else {
                    item.style.display = 'none';
                }
            });
        }

        function resetProgress() {
            if (confirm("Are you sure you want to reset all your course progress?")) {
                courseStatus = {};
                localStorage.removeItem('me_curriculum_status');
                renderCurriculum();
            }
        }

        // Initial Load
        renderCurriculum();
