        let selectedCourseCode = null;

        function buildGrid() {
            const container = document.getElementById('curriculum-container');
            container.innerHTML = '';

            const years = [1, 2, 3, 4];
            years.forEach(yr => {
                const yearSec = document.createElement('div');
                yearSec.className = 'year-section';
                yearSec.innerHTML = `<div class="year-title">YEAR ${yr}</div>`;

                const termsGrid = document.createElement('div');
                termsGrid.className = 'terms-grid';

                for (let t = 1; t <= 4; t++) {
                    const courses = curriculumData.filter(c => c.year === yr && c.term === t);
                    if (courses.length === 0 && t > 2 && yr === 4) continue; // skip empty late terms in Y4

                    const termCol = document.createElement('div');
                    termCol.className = 'term-column';
                    termCol.innerHTML = `<div class="term-header">Term ${t}</div>`;

                    courses.forEach(c => {
                        const card = document.createElement('div');
                        card.className = 'course-card';
                        card.id = `card-${c.code}`;
                        card.onclick = (e) => { e.stopPropagation(); selectCourse(c.code); };

                        card.innerHTML = `
                            <div class="course-code">${c.code}</div>
                            <div class="course-title">${c.title}</div>
                            <div class="course-units">${c.units} Credit Units</div>
                        `;
                        termCol.appendChild(card);
                    });

                    termsGrid.appendChild(termCol);
                }

                yearSec.appendChild(termsGrid);
                container.appendChild(yearSec);
            });
        }

        function selectCourse(code) {
            if (selectedCourseCode === code) {
                selectedCourseCode = null;
            } else {
                selectedCourseCode = code;
            }
            updateGraph();
        }

        document.body.onclick = () => {
            selectedCourseCode = null;
            updateGraph();
        };

        function updateGraph() {
            const svg = document.getElementById('svg-canvas');
            const existingPaths = svg.querySelectorAll('path');
            existingPaths.forEach(p => p.remove());

            const allCards = document.querySelectorAll('.course-card');
            allCards.forEach(c => {
                c.classList.remove('selected', 'is-prereq', 'is-dependent', 'dimmed');
            });

            if (!selectedCourseCode) return;

            const selectedCourse = curriculumData.find(c => c.code === selectedCourseCode);
            if (!selectedCourse) return;

            const selectedCardEl = document.getElementById(`card-${selectedCourseCode}`);
            selectedCardEl.classList.add('selected');

            const prereqCodes = selectedCourse.reqs;
            const dependentCourses = curriculumData.filter(c => c.reqs.includes(selectedCourseCode));

            allCards.forEach(c => {
                const code = c.id.replace('card-', '');
                if (code !== selectedCourseCode && !prereqCodes.includes(code) && !dependentCourses.some(d => d.code === code)) {
                    c.classList.add('dimmed');
                }
            });

            prereqCodes.forEach(pCode => {
                const pEl = document.getElementById(`card-${pCode}`);
                if (pEl) {
                    pEl.classList.add('is-prereq');
                    drawCurve(pEl, selectedCardEl, '#d97706', 'arrow-prereq');
                }
            });

            dependentCourses.forEach(dCourse => {
                const dEl = document.getElementById(`card-${dCourse.code}`);
                if (dEl) {
                    dEl.classList.add('is-dependent');
                    drawCurve(selectedCardEl, dEl, '#2563eb', 'arrow-depends');
                }
            });
        }

        function drawCurve(fromEl, toEl, color, markerId) {
            const svg = document.getElementById('svg-canvas');
            const fromRect = fromEl.getBoundingClientRect();
            const toRect = toEl.getBoundingClientRect();

            const startX = fromRect.right + window.scrollX;
            const startY = fromRect.top + (fromRect.height / 2) + window.scrollY;

            const endX = toRect.left + window.scrollX;
            const endY = toRect.top + (toRect.height / 2) + window.scrollY;

            const dx = Math.abs(endX - startX) * 0.4;
            const pathData = `M ${startX} ${startY} C ${startX + dx} ${startY}, ${endX - dx} ${endY}, ${endX} ${endY}`;

            const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
            path.setAttribute('d', pathData);
            path.setAttribute('stroke', color);
            path.setAttribute('stroke-width', '2.5');
            path.setAttribute('fill', 'none');
            path.setAttribute('marker-end', `url(#${markerId})`);

            svg.appendChild(path);
        }

        window.onresize = updateGraph;
        window.onscroll = updateGraph;

        buildGrid();
    </script>
